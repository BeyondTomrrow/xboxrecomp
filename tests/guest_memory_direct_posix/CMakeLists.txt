cmake_minimum_required(VERSION 3.20)
project(guest_memory_direct_posix C)
add_executable(guest_memory_direct_check check.c ../../src/platform/guest_memory.c)
target_include_directories(guest_memory_direct_check PRIVATE ../../src)
target_compile_options(guest_memory_direct_check PRIVATE -Wall -Wextra)
find_package(Threads REQUIRED)
target_link_libraries(guest_memory_direct_check PRIVATE Threads::Threads)
enable_testing()
add_test(NAME guest_memory_direct COMMAND guest_memory_direct_check)
set_tests_properties(guest_memory_direct PROPERTIES TIMEOUT 30)
